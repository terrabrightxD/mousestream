cmake_minimum_required(VERSION 3.20)
project(MouseStream VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_library(ms_queue STATIC core/queue.c)
add_executable(test_queue tests/test_queue.c)
target_link_libraries(test_queue PRIVATE ms_queue)
# Assertions remain enabled in Release builds.
if(MSVC)
  target_compile_options(test_queue PRIVATE /UNDEBUG /W4 /WX)
  target_compile_options(ms_queue PRIVATE /W4 /WX)
else()
  target_compile_options(test_queue PRIVATE -UNDEBUG -Wall -Wextra -Werror -pedantic)
  target_compile_options(ms_queue PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME queue COMMAND test_queue)
if(WIN32)
  add_library(mousestream SHARED sdk/mousestream.c)
  target_include_directories(mousestream PUBLIC include)
  target_compile_definitions(mousestream PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(mousestream PRIVATE setupapi)
  add_executable(passthrough examples/passthrough.c)
  target_link_libraries(passthrough PRIVATE mousestream)
  if(MSVC)
    target_compile_options(mousestream PRIVATE /W4 /WX)
    target_compile_options(passthrough PRIVATE /W4 /WX)
  endif()
endif()
